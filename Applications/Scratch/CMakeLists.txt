cmake_minimum_required(VERSION 3.28)
project(Scratch LANGUAGES CXX)
include(../../Beam/Config/project.cmake)
beam_configure_project(
  "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}" ENVIRONMENT BEAM_SKIP_CMAKE=1)
include(../../Beam/Config/dependencies.cmake)
include(../../Beam/Config/targets.cmake)
include_directories(${BEAM_INCLUDE_PATH})
include_directories(SYSTEM ${SQLITE_INCLUDE_PATH})
include_directories(SYSTEM ${ZLIB_INCLUDE_PATH})
link_directories(${BOOST_DEBUG_PATH})
link_directories(${BOOST_OPTIMIZED_PATH})
include_directories("${PROJECT_BINARY_DIR}")
set(header_files "${PROJECT_BINARY_DIR}/Version.hpp")
file(GLOB_RECURSE source_files Source/*.cpp)
add_executable(Scratch ${header_files} ${source_files})
beam_configure_target(Scratch DEPENDENCIES
  ASPEN BOOST CRYPTOPP MYSQL OPEN_SSL OPEN_SSL_BASE TCLAP VIPER YAML)
target_compile_definitions(Scratch PRIVATE YAML_CPP_STATIC_DEFINE)
set_source_files_properties(${header_files} PROPERTIES HEADER_FILE_ONLY TRUE)
target_link_libraries(Scratch PUBLIC BeamRoutines)
if(WIN32)
  target_link_libraries(Scratch PUBLIC Crypt32.lib shlwapi)
endif()
beam_install_target(Scratch "${PROJECT_BINARY_DIR}/Application")
beam_configure_clean()
