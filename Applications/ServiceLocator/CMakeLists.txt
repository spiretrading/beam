cmake_minimum_required(VERSION 3.28)
project(ServiceLocator LANGUAGES CXX)
include(../../Beam/Config/project.cmake)
beam_configure_project(
  "${PROJECT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}" ENVIRONMENT BEAM_SKIP_CMAKE=1)
include(../../Beam/Config/dependencies.cmake)
include(../../Beam/Config/targets.cmake)
include_directories(${BEAM_INCLUDE_PATH})
link_directories(${BOOST_DEBUG_PATH})
link_directories(${BOOST_OPTIMIZED_PATH})
include_directories("${PROJECT_BINARY_DIR}")
set(header_files "${PROJECT_BINARY_DIR}/Version.hpp")
file(GLOB_RECURSE source_files Source/*.cpp)
add_executable(ServiceLocator ${header_files} ${source_files})
beam_configure_target(ServiceLocator DEPENDENCIES
  BOOST CRYPTOPP MYSQL OPEN_SSL OPEN_SSL_BASE TCLAP VIPER YAML)
target_compile_definitions(ServiceLocator PRIVATE YAML_CPP_STATIC_DEFINE)
set_source_files_properties(${header_files} PROPERTIES HEADER_FILE_ONLY TRUE)
target_link_libraries(ServiceLocator PUBLIC BeamRoutines)
if(WIN32)
  target_link_libraries(
    ServiceLocator PUBLIC Bcrypt crypt32 Secur32.Lib shlwapi)
endif()
beam_install_target(ServiceLocator "${PROJECT_BINARY_DIR}/Application")
install(FILES Application/config.default.yml Application/setup.py
  ../Python/setup_utils.py DESTINATION "${PROJECT_BINARY_DIR}/Application")
if(UNIX)
  install(PROGRAMS Application/check.sh Application/start.sh
    Application/stop.sh DESTINATION "${PROJECT_BINARY_DIR}/Application")
endif()
beam_configure_clean()
